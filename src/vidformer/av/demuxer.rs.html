9struct IoCtx {
10    canary: u64, // We're doing some unsafe opaque pointer passing, so let's add a canary to make sure we didn't mess up. Valid value is 0xdeadbeef
11    size: u64,
12    reader: Box<dyn crate::io::ReadSeek>,
13    err: Option<std::io::Error>,
14}
15
16unsafe extern "C" fn vidformer_avio_read_packet(
17    opaque: *mut ::std::os::raw::c_void,
18    buf: *mut u8,
19    buf_size: ::std::os::raw::c_int,
20) -> ::std::os::raw::c_int {
21    let io_ctx = &mut *(opaque as *mut IoCtx);
22    debug_assert_eq!(io_ctx.canary, 0xdeadbeef);
23
24    if buf_size < 0 {
25        return ffi::AVERROR_EXTERNAL;
26    }
27    let buf: &mut [u8] = unsafe { slice::from_raw_parts_mut(buf, buf_size as usize) };
28    let read = io_ctx.reader.read(buf);
29    match read {
30        Ok(read) => {
31            if read == 0 {
32                ffi::AVERROR_EOF
33            } else {
34                read as i32
35            }
36        }
37        Err(e) => {
38            error!("Error reading packet: {}", e);
39            if io_ctx.err.is_none() {
40                io_ctx.err = Some(e);
41            }
42            ffi::AVERROR_EXTERNAL
43        }
44    }
45}

47unsafe extern "C" fn vidformer_avio_seek(
48    opaque: *mut ::std::os::raw::c_void,
49    offset: i64,
50    whence: ::std::os::raw::c_int,
51) -> i64 {
52    let io_ctx = &mut *(opaque as *mut IoCtx);
53    debug_assert_eq!(io_ctx.canary, 0xdeadbeef);
54
55    // Never panic here — unwinding across the C ABI aborts the process.
56    let whence_raw = whence as u32;
57    if whence_raw & ffi::AVSEEK_SIZE != 0 {
58        // libav way of asking for the size of the file
59        return io_ctx.size as i64;
60    }
61    let whence = match whence_raw & !ffi::AVSEEK_FORCE {
62        ffi::SEEK_CUR => std::io::SeekFrom::Current(offset),
63        ffi::SEEK_END => std::io::SeekFrom::End(offset),
64        ffi::SEEK_SET => std::io::SeekFrom::Start(offset as u64),
65        _ => {
66            error!("Unsupported seek whence ({})", whence);
67            return ffi::AVERROR_EXTERNAL as i64;
68        }
69    };
70
71    let seeked = io_ctx.reader.seek(whence);
72    match seeked {
73        Ok(seeked) => seeked as i64,
74        Err(e) => {
75            error!("Error seeking: {}", e);
76            if io_ctx.err.is_none() {
77                io_ctx.err = Some(e);
78            }
79            ffi::AVERROR_EXTERNAL as i64
80        }
81    }
82}

84pub struct Demuxer {
85    pub format_context: *mut ffi::AVFormatContext,
86    avio_context: *mut ffi::AVIOContext,
87    #[allow(unused)] // We need to keep this alive since libav keeps it as an opaque pointer
88    io_ctx: std::pin::Pin<std::boxed::Box<IoCtx>>,
89    pub time_base: Rational64,
90    pub codec: *const ffi::AVCodec,
91    pub codec_parameters: *const ffi::AVCodecParameters,
92    pub video_stream_index: Option<usize>,
93    pub stream: *mut ffi::AVStream,
94}
95
96impl Demuxer {
97    pub fn new(
98        file_path: &str,
99        stream_idx: usize,
100        service: &crate::service::Service,
101        file_size: u64,
102        io_runtime_handle: &tokio::runtime::Handle,
103        io_cache: Option<(&dyn crate::io::IoWrapper, &str)>,
104    ) -> Result<Self, crate::Error> {
105        let format_context = unsafe { ffi::avformat_alloc_context() };
106        if format_context.is_null() {
107            return Err(crate::Error::AVError(
108                "could not allocate memory for Format Context".to_string(),
109            ));
110        }
111
112        debug!("Opening {} for read", file_path);
113
114        let op = service.blocking_operator(io_runtime_handle)?;
115
116        let reader: opendal::BlockingReader = op.reader(file_path).map_err(|e| {
117            if e.kind() == opendal::ErrorKind::NotFound {
118                crate::Error::IOError(format!("File `{}` not found", file_path))
119            } else {
120                crate::Error::IOError(format!("OpenDAL error: {}", e))
121            }
122        })?;
123
124        let reader: opendal::StdReader = match reader.into_std_read(0..file_size) {
125            Ok(reader) => reader,
126            Err(err) => {
127                return Err(crate::Error::IOError(format!(
128                    "OpenDAL failed to convert BlockingReader to StdReader: {}",
129                    err
130                )));
131            }
132        };
133
134        let reader = match io_cache {
135            Some((io_wrapper, fuid)) => io_wrapper.wrap(Box::new(reader), fuid),
136            None => Box::new(std::io::BufReader::with_capacity(128 * 1024, reader)),
137        };
138
139        let io_ctx = IoCtx {
140            canary: 0xdeadbeef,
141            size: file_size,
142            reader: Box::new(reader),
143            err: None,
144        };
145        let io_ctx = Box::pin(io_ctx);
146        let io_ctx_ptr =
147            io_ctx.as_ref().get_ref() as *const IoCtx as *mut IoCtx as *mut ::std::os::raw::c_void;
148
149        let avio_buffer_size = 16 * 1024;
150        let avio_buffer: *mut std::ffi::c_void =
151            unsafe { ffi::av_malloc(avio_buffer_size as usize) };
152
153        let avio_context = unsafe {
154            ffi::avio_alloc_context(
155                avio_buffer as *mut u8,
156                avio_buffer_size,
157                0,
158                io_ctx_ptr,
159                Some(vidformer_avio_read_packet),
160                None,
161                Some(vidformer_avio_seek),
162            )
163        };
164
165        if avio_context.is_null() {
166            return Err(crate::Error::AVError(
167                "could not allocate memory for AVIO Context".to_string(),
168            ));
169        }
170
171        unsafe {
172            (*format_context).pb = avio_context;
173        }
174
175        // `avformat_open_input` frees the format context on failure but never
176        // touches `pb`, which libav has grown up to `probesize` while probing.
177        let mut demuxer = Demuxer {
178            format_context,
179            avio_context,
180            io_ctx,
181            time_base: Rational64::new(0, 1),
182            codec: ptr::null(),
183            codec_parameters: ptr::null(),
184            video_stream_index: None,
185            stream: ptr::null_mut(),
186        };
187
188        let ret = unsafe {
189            ffi::avformat_open_input(
190                &mut demuxer.format_context,
191                ptr::null_mut(),
192                ptr::null_mut(),
193                ptr::null_mut(),
194            )
195        };
196        if ret != 0 {
197            if ret == ffi::AVERROR_EXTERNAL {
198                let err = match demuxer.io_ctx.err.as_ref() {
199                    Some(err) => err,
200                    None => {
201                        return Err(crate::Error::IOError(
202                            "IO error while opening media".to_string(),
203                        ))
204                    }
205                };
206                if err.kind() == std::io::ErrorKind::NotFound {
207                    return Err(crate::Error::IOError(format!(
208                        "File `{}` not found",
209                        file_path
210                    )));
211                } else {
212                    return Err(crate::Error::IOError(format!("OpenDAL error: {}", err)));
213                }
214            }
215
216            let err_str = crate::util::av_strerror(ret);
217            return Err(crate::Error::AVError(format!(
218                "failed to open media format: {}",
219                err_str
220            )));
221        }
222
223        // TODO: This may decode a few frames, which could be slow. Maybe don't do that when actually running a spec?
224        if unsafe { ffi::avformat_find_stream_info(demuxer.format_context, ptr::null_mut()) } < 0 {
225            return Err(crate::Error::AVError(
226                "could not get the stream info".to_string(),
227            ));
228        }
229
230        let mut codec_ptr: *const ffi::AVCodec = ptr::null_mut();
231        let mut codec_parameters_ptr: *const ffi::AVCodecParameters = ptr::null_mut();
232        let mut video_stream_index = None;
233
234        // libav leaves `streams` null for a container with no streams at all,
235        // and `from_raw_parts_mut` requires non-null even at length 0.
236        let nb_streams = unsafe { (*demuxer.format_context).nb_streams as usize };
237        if stream_idx >= nb_streams {
238            return Err(crate::Error::AVError(format!(
239                "Stream index {} out of range (file has {} streams)",
240                stream_idx, nb_streams
241            )));
242        }
243
244        let streams =
245            unsafe { slice::from_raw_parts_mut((*demuxer.format_context).streams, nb_streams) };
246
247        for (i, stream) in streams
248            .iter_mut()
249            .map(|stream| unsafe { stream.as_mut() }.expect("null stream pointer"))
250            .enumerate()
251        {
252            if i != stream_idx {
253                stream.discard = ffi::AVDiscard_AVDISCARD_ALL;
254                continue;
255            }
256
257            let local_codec_params = unsafe { stream.codecpar.as_ref() }.expect("codecpar is null");
258            let local_codec =
259                match unsafe { ffi::avcodec_find_decoder(local_codec_params.codec_id).as_ref() } {
260                    Some(codec) => codec,
261                    None => return Err(crate::Error::AVError("Unsupported codec".to_string())),
262                };
263
264            if local_codec_params.codec_type == ffi::AVMediaType_AVMEDIA_TYPE_VIDEO {
265                if video_stream_index.is_none() {
266                    video_stream_index = Some(i);
267                    codec_ptr = local_codec;
268                    codec_parameters_ptr = local_codec_params;
269                }
270            } else {
271                // The requested stream is not a video stream
272                let codec_type = match local_codec_params.codec_type {
273                    ffi::AVMediaType_AVMEDIA_TYPE_AUDIO => "audio",
274                    ffi::AVMediaType_AVMEDIA_TYPE_SUBTITLE => "subtitle",
275                    ffi::AVMediaType_AVMEDIA_TYPE_DATA => "data",
276                    ffi::AVMediaType_AVMEDIA_TYPE_ATTACHMENT => "attachment",
277                    _ => "unknown",
278                };
279                return Err(crate::Error::AVError(format!(
280                    "Stream {} is not a video stream (it is a {} stream). Please specify a valid video stream index.",
281                    i, codec_type
282                )));
283            }
284        }
285
286        let time_base = crate::util::avrat_to_rat(&unsafe { (*streams[stream_idx]).time_base })?;
287        debug_assert!(!time_base.is_zero());
288
289        demuxer.time_base = time_base;
290        demuxer.codec = codec_ptr;
291        demuxer.codec_parameters = codec_parameters_ptr;
292        demuxer.video_stream_index = video_stream_index;
293        demuxer.stream = unsafe { streams[stream_idx].as_mut() }.unwrap();
294        Ok(demuxer)
295    }

297    pub fn seek(&mut self, ts: &Rational64) -> Result<(), crate::Error> {
298        let seek_ts = ts / self.time_base;
299        assert_eq!(*seek_ts.denom(), 1);
300        let seek_ts = *seek_ts.numer();
301
302        debug!("Seeking to {}", ts);
303
304        if unsafe { ffi::avformat_seek_file(self.format_context, 0, seek_ts, seek_ts, seek_ts, 0) }
305            < 0
306        {
307            return Err(crate::Error::AVError("failed to seek file".to_string()));
308        }
309
310        Ok(())
311    }

313    pub fn read_packet(&mut self, packet: *mut ffi::AVPacket) -> Option<()> {
314        loop {
315            if unsafe { ffi::av_read_frame(self.format_context, packet) } >= 0 {
316                // debug!(
317                //     "DEMUX - Packet stream index {} pts = {} dts = {} key_frame = {} size = {}",
318                //     unsafe { (*packet).stream_index },
319                //     unsafe { (*packet).pts },
320                //     unsafe { (*packet).dts },
321                //     unsafe { (*packet).flags } as u32 & ffi::AV_PKT_FLAG_KEY != 0,
322                //     unsafe { (*packet).size }
323                // );
324
325                // Is this packet from the correct stream?
326                if unsafe { (*packet).stream_index } as usize == self.video_stream_index.unwrap() {
327                    // If so, we have updated the pointer, so return
328                    return Some(());
329                } else {
330                    // If not, unref the packet and keep trying
331                    unsafe {
332                        ffi::av_packet_unref(packet);
333                    }
334                    continue;
335                }
336            } else {
337                return None;
338            }
339        }
340    }
341
342    pub fn close(&mut self) {
343        self.free_resources();
344    }
345
346    /// Idempotent, so `close()` and then `Drop` is safe.
347    fn free_resources(&mut self) {
348        unsafe {
349            if !self.format_context.is_null() {
350                // The real field, so libav nulls it.
351                ffi::avformat_close_input(&mut self.format_context);
352            }
353
354            if !self.avio_context.is_null() {
355                ffi::av_freep(
356                    &mut (*self.avio_context).buffer as *mut *mut u8
357                        as *mut *mut std::os::raw::c_void
358                        as *mut std::os::raw::c_void,
359                );
360                ffi::av_free(self.avio_context as *mut std::os::raw::c_void);
361                self.avio_context = std::ptr::null_mut();
362            }
363        }
364    }
365}

387            0,

393        let io_runtime = tokio::runtime::Builder::new_multi_thread()
394            .worker_threads(1)
395            .enable_all()
396            .build()
397            .unwrap();

400            TEST_VID,
401            TEST_STREAM,
402            &service,
403            profile.file_size,
404            io_runtime.handle(),
405            None,
406        )
407        .unwrap();
408
409        let packet = unsafe { ffi::av_packet_alloc().as_mut() }
410            .expect("failed to allocated memory for AVPacket");

414            demuxed_packets += 1;
415        }
416        demuxer.close();
417
